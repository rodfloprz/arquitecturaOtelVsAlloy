### 3. Alloy: recolección de logs con WAL

```alloy
local.file_match "app_logs" {
  path_targets = [{
    __path__    = "/var/log/app/*.log",
    job         = "app",
    environment = sys.env("OBS_ENVIRONMENT"),
  }]
}

loki.source.file "app" {
  targets    = local.file_match.app_logs.targets
  forward_to = [loki.process.app.receiver]
}

// Redacción y parsing; mantener lo mínimo para no acoplar el formato a Grafana.
loki.process "app" {
  stage.json { expressions = { level = "level", trace_id = "trace_id" } }
  stage.labels { values = { level = "" } }
  forward_to = [loki.write.cloud.receiver]
}

loki.write "cloud" {
  endpoint {
    url = sys.env("GCLOUD_LOKI_URL")
    basic_auth {
      username = sys.env("GCLOUD_LOKI_USER")
      password = sys.env("GCLOUD_RW_API_KEY")
    }
  }
  // Persistencia en disco para cortes largos de Loki.
  wal { enabled = true }
}
```

### 4. Contrib: recolección de logs con filelog y cola persistente

```yaml
receivers:
  filelog:
    include: [/var/log/app/*.log]
    start_at: end
    storage: file_storage/pos         # posiciones persistentes entre reinicios
    operators:
      - type: json_parser
processors:
  memory_limiter: { check_interval: 1s, limit_percentage: 80 }
  batch: {}
extensions:
  file_storage/pos: { directory: /var/lib/otelcol/pos }
  file_storage/queue: { directory: /var/lib/otelcol/queue }
  basicauth/grafana:
    client_auth: { username: "${env:GCLOUD_OTLP_USER}", password: "${env:GCLOUD_RW_API_KEY}" }
exporters:
  otlphttp/grafana:
    endpoint: ${env:GCLOUD_OTLP_URL}
    auth: { authenticator: basicauth/grafana }
    sending_queue: { enabled: true, storage: file_storage/queue }
service:
  extensions: [file_storage/pos, file_storage/queue, basicauth/grafana]
  pipelines:
    logs: { receivers: [filelog], processors: [memory_limiter, batch], exporters: [otlphttp/grafana] }
```

### 5. Alloy: recepción y procesamiento de trazas (agente local)

```alloy
otelcol.receiver.otlp "in" {
  grpc { endpoint = "0.0.0.0:4317" }
  http { endpoint = "0.0.0.0:4318" }
  output { traces = [otelcol.processor.memory_limiter.default.input] }
}

otelcol.processor.memory_limiter "default" {
  check_interval = "1s"
  limit          = "400MiB"
  output { traces = [otelcol.processor.resourcedetection.default.input] }
}

otelcol.processor.resourcedetection "default" {
  detectors = ["env", "system"]
  output { traces = [otelcol.processor.batch.default.input] }
}

otelcol.processor.batch "default" {
  output { traces = [otelcol.exporter.loadbalancing.tier1.input] }
}
```

### 6. Contrib: recepción y procesamiento de trazas (agente local)

```yaml
receivers:
  otlp:
    protocols: { grpc: { endpoint: 0.0.0.0:4317 }, http: { endpoint: 0.0.0.0:4318 } }
processors:
  memory_limiter: { check_interval: 1s, limit_percentage: 80, spike_limit_percentage: 20 }
  resourcedetection: { detectors: [env, system], timeout: 2s }
  batch: {}
exporters:
  load_balancing:
    routing_key: traceID
    protocol: { otlp: { tls: { insecure: false }, timeout: 5s } }
    resolver:
      dns: { hostname: trace-sampler-headless.observability.svc.cluster.local, port: "4317" }
service:
  pipelines:
    traces:
      receivers: [otlp]
      processors: [memory_limiter, resourcedetection, batch]
      exporters: [load_balancing]
```

### 7. Contrib: nivel 1 con load_balancing y resiliencia en el exportador principal

```yaml
exporters:
  load_balancing:
    routing_key: traceID
    timeout: 10s
    retry_on_failure: { enabled: true, initial_interval: 5s, max_interval: 30s, max_elapsed_time: 300s }
    sending_queue: { enabled: true, num_consumers: 2, queue_size: 1000 }
    protocol: { otlp: { timeout: 1s, sending_queue: { enabled: true } } }
    resolver:
      k8s:
        service: trace-sampler.observability     # Service headless apuntando a un StatefulSet
        ports: [4317]
        return_hostnames: true
```

Requiere RBAC `get`, `list` y `watch` sobre EndpointSlice para el service account del collector {{V}}.

### 8. Alloy: patrón equivalente de balanceo y tail sampling

```alloy
otelcol.exporter.loadbalancing "tier1" {
  routing_key = "traceID"
  resolver {
    kubernetes { service = "trace-sampler.observability" }
  }
  protocol { otlp { client { tls { insecure = false } } } }
}

// En el nivel 2 (StatefulSet):
otelcol.processor.tail_sampling "main" {
  decision_wait = "30s"
  num_traces    = 100000
  decision_cache = {
    sampled_cache_size     = 200000,
    non_sampled_cache_size = 200000,
  }
  policy {
    name = "errors"
    type = "status_code"
    status_code { status_codes = ["ERROR"] }
  }
  policy {
    name = "slow"
    type = "latency"
    latency { threshold_ms = 2000 }
  }
  policy {
    name = "base"
    type = "probabilistic"
    probabilistic { sampling_percentage = 10 }
  }
  output { traces = [otelcol.exporter.otlphttp.cloud.input] }
}
```

Brecha: Alloy no tiene `groupbytrace`; se compensa con `decision_wait` y `decision_cache`. Los números de `num_traces` y de caché son ilustrativos y salen de las variables del dimensionamiento.
